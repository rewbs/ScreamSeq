cmake_minimum_required(VERSION 3.24)
project(ScreamSeqWorkspaceTests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Workspace persistence tests require Windows")
endif()
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(workspace-layout-tests WorkspaceLayoutsTests.cpp)
target_compile_features(workspace-layout-tests PRIVATE cxx_std_20)
target_include_directories(workspace-layout-tests PRIVATE "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(workspace-layout-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(workspace-layout-tests PRIVATE /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
endif()
add_test(NAME workspace-layout-tests COMMAND workspace-layout-tests)
set_tests_properties(workspace-layout-tests PROPERTIES LABELS "native-ui;preferences" TIMEOUT 30)
add_executable(native-tool-window-tests NativeToolWindowTests.cpp)
target_compile_features(native-tool-window-tests PRIVATE cxx_std_20)
target_include_directories(native-tool-window-tests PRIVATE "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(native-tool-window-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(native-tool-window-tests PRIVATE /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
endif()
target_link_libraries(native-tool-window-tests PRIVATE d3d11 d2d1 dwrite dxgi dwmapi comctl32)
add_test(NAME native-tool-window-tests COMMAND native-tool-window-tests)
set_tests_properties(native-tool-window-tests PROPERTIES LABELS "native-ui;private-desktop" TIMEOUT 30)
