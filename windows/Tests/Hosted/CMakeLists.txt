cmake_minimum_required(VERSION 3.24)
project(ScreamSeqHostedTests LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_EDITOR_BUILD "${ROOT}/bin/windows-editor" CACHE PATH "Existing ARM64 shared engine build")
if(NOT TARGET TrackerEditor)
  foreach(lib TrackerEditor OpenMPTCore TrackerFLAC)
    add_library(${lib} STATIC IMPORTED GLOBAL)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${SCREAMSEQ_EDITOR_BUILD}/Release/${lib}.lib")
  endforeach()
  target_link_libraries(TrackerEditor INTERFACE OpenMPTCore)
  target_link_libraries(OpenMPTCore INTERFACE TrackerFLAC shlwapi ole32 rpcrt4)
  target_include_directories(OpenMPTCore INTERFACE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include"
    "${ROOT}/include/miniz" "${ROOT}/include/minimp3" "${ROOT}/include/stb_vorbis" "${ROOT}/include/flac/include")
  target_compile_definitions(OpenMPTCore INTERFACE LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE
    MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL
    NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
  target_compile_options(OpenMPTCore INTERFACE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
endif()
# Reuse the application's target when included by a parent, or build it here.
if(NOT TARGET TrackerHosted)
  add_subdirectory("${ROOT}/editor/hosted" "${CMAKE_CURRENT_BINARY_DIR}/hosted")
endif()
add_executable(hosted-smoke HostedSmoke.cpp)
target_link_libraries(hosted-smoke PRIVATE TrackerHosted)
target_compile_options(hosted-smoke PRIVATE "/FI${ROOT}/windows/cmake/WindowsTestPreamble.hpp")
target_link_options(hosted-smoke PRIVATE /STACK:8388608)
add_test(NAME hosted-smoke COMMAND hosted-smoke)
set_tests_properties(hosted-smoke PROPERTIES LABELS "hosted;functional;no-realtime-audit")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/prepare_tests.py"
  "${ROOT}/mac/Tests/SignalRoutingTests.mm" "${ROOT}/mac/Tests/MusicalAutomationTests.mm"
  "${ROOT}/mac/Tests/PreciseNoteTests.mm" "${ROOT}/mac/Tests/NativeSignalGraphTests.mm")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_tests.py"
  "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(test SignalRouting MusicalAutomation PreciseNotes NativeSignalGraph)
  add_executable(hosted-${test} "${CMAKE_CURRENT_BINARY_DIR}/generated/${test}.cpp")
  if(NOT test STREQUAL "SignalRouting")
    # Explicitly bound test provider: this is NOT a VST3 loader qualification.
    target_sources(hosted-${test} PRIVATE FixtureBackend.cpp)
  endif()
  target_link_libraries(hosted-${test} PRIVATE TrackerHosted)
  target_include_directories(hosted-${test} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(hosted-${test} PRIVATE "/FI${ROOT}/windows/cmake/WindowsTestPreamble.hpp")
  target_link_options(hosted-${test} PRIVATE /STACK:8388608)
  add_test(NAME hosted-${test} COMMAND hosted-${test})
  set_tests_properties(hosted-${test} PROPERTIES LABELS "hosted;functional;no-realtime-audit")
endforeach()
foreach(test BackendContract BuiltinGraph)
  add_executable(hosted-${test} "${test}.cpp")
  if(test STREQUAL "BackendContract")
    target_sources(hosted-${test} PRIVATE FixtureBackend.cpp)
  endif()
  target_link_libraries(hosted-${test} PRIVATE TrackerHosted)
  target_compile_options(hosted-${test} PRIVATE "/FI${ROOT}/windows/cmake/WindowsTestPreamble.hpp")
  target_link_options(hosted-${test} PRIVATE /STACK:8388608)
  add_test(NAME hosted-${test} COMMAND hosted-${test})
  set_tests_properties(hosted-${test} PROPERTIES LABELS "hosted;functional;no-realtime-audit")
endforeach()
