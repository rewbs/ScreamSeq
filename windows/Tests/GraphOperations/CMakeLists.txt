cmake_minimum_required(VERSION 3.24)
project(ScreamSeqGraphOperationsTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_ENGINE_LIB_DIR "${ROOT}/bin/windows-editor/Release" CACHE PATH "Existing matching engine libraries")
add_executable(graph-operations-tests GraphOperationsTests.cpp ../../Session/GraphOperations.cpp ../../Project/NativeMetadata.cpp)
target_include_directories(graph-operations-tests PRIVATE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/flac/include" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(graph-operations-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL)
target_compile_options(graph-operations-tests PRIVATE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_options(graph-operations-tests PRIVATE /STACK:8388608)
target_link_libraries(graph-operations-tests PRIVATE "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerEditor.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/OpenMPTCore.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerFLAC.lib" shlwapi ole32 rpcrt4)

foreach(SCENARIO createReadHistory nodesAndCloning automationAndBanks assignmentsRoutesLayoutCommands hostHooks strictValidation dryRunsAndRedo callbackOrderingAndUnrelatedData)
  add_test(NAME graph-${SCENARIO} COMMAND graph-operations-tests ${SCENARIO})
