cmake_minimum_required(VERSION 3.21)
project(ScreamSeqApiTests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Named pipe tests require Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(ApiTests ApiTests.cpp)
target_include_directories(ApiTests PRIVATE "${ROOT}/windows/Api" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(ApiTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ApiTests PRIVATE advapi32)
target_sources(ApiTests PRIVATE "${ROOT}/windows/Api/PipeServer.cpp")
enable_testing()
add_test(NAME ApiTests COMMAND ApiTests)
set_tests_properties(ApiTests PROPERTIES TIMEOUT 30)
add_executable(SessionCacheTests SessionCacheTests.cpp)
target_include_directories(SessionCacheTests PRIVATE "${ROOT}/windows/Api" "${ROOT}/include/nlohmann-json/include")
add_test(NAME SessionCacheTests COMMAND SessionCacheTests)
set_tests_properties(SessionCacheTests PROPERTIES TIMEOUT 120)
