cmake_minimum_required(VERSION 3.24)
project(ScreamSeqAssetTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CORE_LIBRARY_DIR "${ROOT}/bin/windows-editor/Release" CACHE PATH "Built shared engine/editor libraries")
# Compile changed shared import units, never prove a fix with a stale archive.
add_executable(asset-tests AssetTests.cpp ImportTests.cpp VorbisProbeTests.cpp
  "${ROOT}/editor/TrackerDocument.cpp" "${ROOT}/editor/SampleImport.cpp"
  "${ROOT}/windows/Project/NativeMetadata.cpp")
if(EXISTS "${ROOT}/windows/Session/AssetOperations.cpp")
  target_sources(asset-tests PRIVATE "${ROOT}/windows/Session/AssetOperations.cpp")
  target_compile_definitions(asset-tests PRIVATE HAVE_ASSET_OPERATIONS)
endif()
target_include_directories(asset-tests PRIVATE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/nlohmann-json/include" "${ROOT}/include/flac/include")
target_compile_definitions(asset-tests PRIVATE ASSET_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/Fixtures")
target_compile_definitions(asset-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL)
target_compile_options(asset-tests PRIVATE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_options(asset-tests PRIVATE /STACK:8388608 /MAP)
target_link_libraries(asset-tests PRIVATE "${CORE_LIBRARY_DIR}/TrackerEditor.lib" "${CORE_LIBRARY_DIR}/OpenMPTCore.lib" "${CORE_LIBRARY_DIR}/TrackerFLAC.lib" shlwapi ole32 rpcrt4)
enable_testing()
add_test(NAME asset-tests COMMAND asset-tests)
add_test(NAME asset-import-tests COMMAND asset-tests --imports)
add_test(NAME vorbis-probe-tests COMMAND asset-tests --probe-vorbis)
add_test(NAME mo3-vorbis-probe-tests COMMAND asset-tests --probe-mo3-vorbis)
add_test(NAME shared-external-import-tests COMMAND asset-tests --probe-external-import)
add_test(NAME api-external-import-tests COMMAND asset-tests --external-imports)
set_tests_properties(asset-tests asset-import-tests vorbis-probe-tests mo3-vorbis-probe-tests
  shared-external-import-tests api-external-import-tests PROPERTIES TIMEOUT 60)
