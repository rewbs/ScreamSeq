cmake_minimum_required(VERSION 3.21)
project(ScreamSeqBinaryPlistTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(BinaryPlistTests BinaryPlistTests.cpp "${ROOT}/windows/Project/BinaryPlist.cpp")
target_compile_features(BinaryPlistTests PRIVATE cxx_std_20)
target_include_directories(BinaryPlistTests PRIVATE "${ROOT}/windows/Project" "${ROOT}/include/nlohmann-json/include")
if(MSVC)
  target_compile_options(BinaryPlistTests PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
endif()
enable_testing()
add_test(NAME BinaryPlistTests COMMAND BinaryPlistTests)
set_tests_properties(BinaryPlistTests PROPERTIES TIMEOUT 60)
