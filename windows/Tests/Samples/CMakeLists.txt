cmake_minimum_required(VERSION 3.24)
project(ScreamSeqSampleLibraryTests LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(sample-library-index-tests LibraryIndexTests.cpp "${ROOT}/windows/Samples/LibraryIndex.cpp")
target_compile_features(sample-library-index-tests PRIVATE cxx_std_20)
target_include_directories(sample-library-index-tests PRIVATE "${ROOT}" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(sample-library-index-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(sample-library-index-tests PRIVATE /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_libraries(sample-library-index-tests PRIVATE bcrypt normaliz)
add_test(NAME sample-library-index COMMAND sample-library-index-tests "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(sample-library-index PROPERTIES TIMEOUT 30)
add_executable(sample-library-tests LibraryTests.cpp "${ROOT}/windows/Samples/Library.cpp" "${ROOT}/windows/Samples/LibraryIndex.cpp")
target_compile_features(sample-library-tests PRIVATE cxx_std_20)
target_include_directories(sample-library-tests PRIVATE "${ROOT}" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(sample-library-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(sample-library-tests PRIVATE /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_libraries(sample-library-tests PRIVATE bcrypt normaliz ole32)
add_test(NAME sample-library-worker COMMAND sample-library-tests "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(sample-library-worker PROPERTIES TIMEOUT 30)
