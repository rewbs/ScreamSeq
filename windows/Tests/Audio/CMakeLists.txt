cmake_minimum_required(VERSION 3.24)
project(ScreamSeqWasapiDiagnostics LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "WASAPI diagnostics require Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(wasapi-diagnostic WasapiDeviceTests.cpp ../../Audio/WasapiDevice.cpp)
target_compile_definitions(wasapi-diagnostic PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
option(SCREAMSEQ_WASAPI_ALLOCATION_AUDIT "Audit host C++ new/delete in the render service scope" ON)
if(SCREAMSEQ_WASAPI_ALLOCATION_AUDIT)
  target_compile_definitions(wasapi-diagnostic PRIVATE SCREAMSEQ_WASAPI_ALLOCATION_AUDIT=1)
endif()
target_link_libraries(wasapi-diagnostic PRIVATE ole32 avrt uuid)
if(MSVC)
  target_compile_options(wasapi-diagnostic PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_test(NAME wasapi-lifecycle COMMAND wasapi-diagnostic)
# Real-device tests are deliberately opt-in, not part of unattended CTest.
