cmake_minimum_required(VERSION 3.24)
project(ScreamSeqHostedProjectTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_EDITOR_BUILD "${ROOT}/bin/windows-snapshot-fix" CACHE PATH "Existing native ARM64 core build")
foreach(lib TrackerEditor OpenMPTCore TrackerFLAC)
  add_library(${lib} STATIC IMPORTED GLOBAL)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${SCREAMSEQ_EDITOR_BUILD}/Release/${lib}.lib")
endforeach()
target_link_libraries(TrackerEditor INTERFACE OpenMPTCore)
target_link_libraries(OpenMPTCore INTERFACE TrackerFLAC shlwapi ole32 rpcrt4)
target_include_directories(OpenMPTCore INTERFACE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/flac/include" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(OpenMPTCore INTERFACE LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
target_compile_options(OpenMPTCore INTERFACE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
add_subdirectory("${ROOT}/editor/hosted" "${CMAKE_CURRENT_BINARY_DIR}/hosted")
add_executable(hosted-project-tests HostedProjectTests.cpp CallbackAudit.cpp
  "${ROOT}/windows/Session/HostedProject.cpp"
  "${ROOT}/windows/Project/NativeProject.cpp" "${ROOT}/windows/Project/NativeMetadata.cpp" "${ROOT}/windows/Project/BinaryPlist.cpp")
target_link_libraries(hosted-project-tests PRIVATE TrackerHosted)
target_compile_definitions(hosted-project-tests PRIVATE SCREAMSEQ_WASAPI_ALLOCATION_AUDIT=1)
target_link_options(hosted-project-tests PRIVATE /STACK:8388608)
enable_testing()
add_test(NAME hosted-project-tests COMMAND hosted-project-tests)
