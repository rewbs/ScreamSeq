cmake_minimum_required(VERSION 3.24)
project(ScreamSeqMetadataTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_ENGINE_LIB_DIR "${ROOT}/bin/windows-workspace/Release" CACHE PATH "Existing ARM64 engine libraries")
add_executable(native-metadata-tests ../ProjectNative/NativeMetadataTests.cpp ../../Project/NativeMetadata.cpp)
target_include_directories(native-metadata-tests PRIVATE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/flac/include" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(native-metadata-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL)
target_compile_options(native-metadata-tests PRIVATE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_options(native-metadata-tests PRIVATE /STACK:8388608)
target_link_libraries(native-metadata-tests PRIVATE "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerEditor.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/OpenMPTCore.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerFLAC.lib" shlwapi ole32 rpcrt4)
enable_testing()
add_test(NAME native-metadata-tests COMMAND native-metadata-tests)
set(SCREAMSEQ_METADATA_FIXTURE "" CACHE FILEPATH "Opt-in JSON metadata exported from the hash-pinned Mac fixture")
if(SCREAMSEQ_METADATA_FIXTURE)
  if(NOT EXISTS "${SCREAMSEQ_METADATA_FIXTURE}")
    message(FATAL_ERROR "SCREAMSEQ_METADATA_FIXTURE does not exist")
  endif()
  add_test(NAME native-metadata-mac-fixture COMMAND native-metadata-tests "${SCREAMSEQ_METADATA_FIXTURE}")
endif()
