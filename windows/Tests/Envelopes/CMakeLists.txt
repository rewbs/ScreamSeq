cmake_minimum_required(VERSION 3.24)
project(ScreamSeqEnvelopeTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_ENGINE_LIB_DIR "${ROOT}/bin/windows-editor/Release" CACHE PATH "Existing matching ARM64 core libraries")
add_executable(envelope-tests EnvelopeTests.cpp "${ROOT}/windows/Session/EnvelopeOperations.cpp")
target_sources(envelope-tests PRIVATE "${ROOT}/windows/Project/BinaryPlist.cpp" "${ROOT}/windows/Project/NativeMetadata.cpp")
target_include_directories(envelope-tests PRIVATE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/flac/include" "${ROOT}/include/nlohmann-json/include")
target_compile_definitions(envelope-tests PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL)
target_compile_options(envelope-tests PRIVATE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
target_link_options(envelope-tests PRIVATE /STACK:8388608)
target_link_libraries(envelope-tests PRIVATE "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerEditor.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/OpenMPTCore.lib" "${SCREAMSEQ_ENGINE_LIB_DIR}/TrackerFLAC.lib" shlwapi ole32 rpcrt4 bcrypt)
enable_testing()
add_test(NAME envelope-instrumentMetadataBudget COMMAND envelope-tests instrumentMetadataBudget)
foreach(SCENARIO bankHistory linkedUses catalogueCopies strictShapesAndTargets parameterHooks instrumentBaking refitAndIndependentCopies capacityBeforePlayback catalogueValidation catalogueWriters bankPersistence jsonRead jsonImport jsonPublish jsonDefault jsonStrict jsonLimits jsonOutputBudget jsonGuards plistWriters jsonResponseBudget)
  add_test(NAME envelope-${SCENARIO} COMMAND envelope-tests ${SCENARIO})
  set_tests_properties(envelope-${SCENARIO} PROPERTIES TIMEOUT 60)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME envelope-catalogueOracle COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/catalogue_oracle.py" "$<TARGET_FILE:envelope-tests>")
set_tests_properties(envelope-catalogueOracle PROPERTIES TIMEOUT 90)
