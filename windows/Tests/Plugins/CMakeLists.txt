cmake_minimum_required(VERSION 3.24)
project(ScreamSeqVST3Qualification LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCREAMSEQ_EDITOR_BUILD "${ROOT}/bin/windows-snapshot-fix" CACHE PATH "Shared engine build")
foreach(lib TrackerEditor OpenMPTCore TrackerFLAC)
  add_library(${lib} STATIC IMPORTED GLOBAL)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${SCREAMSEQ_EDITOR_BUILD}/Release/${lib}.lib")
endforeach()
target_link_libraries(TrackerEditor INTERFACE OpenMPTCore)
target_link_libraries(OpenMPTCore INTERFACE TrackerFLAC shlwapi ole32 rpcrt4)
target_include_directories(OpenMPTCore INTERFACE "${ROOT}" "${ROOT}/src" "${ROOT}/common" "${ROOT}/include" "${ROOT}/include/miniz" "${ROOT}/include/minimp3" "${ROOT}/include/stb_vorbis" "${ROOT}/include/flac/include")
target_compile_definitions(OpenMPTCore INTERFACE LIBOPENMPT_BUILD OPENMPT_EDITOR_CORE MPT_WITH_MINIZ MPT_WITH_MINIMP3 MPT_WITH_STBVORBIS MPT_WITH_FLAC FLAC__NO_DLL NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
target_compile_options(OpenMPTCore INTERFACE /utf-8 /bigobj /permissive- /Zc:__cplusplus /Zc:preprocessor)
include("${ROOT}/windows/Plugins/Provider.cmake")
add_subdirectory("${ROOT}/editor/hosted" hosted)
screamseq_windows_vst3_target(TrackerHosted)
add_executable(vst3-tests VST3Tests.cpp)
target_link_libraries(vst3-tests PRIVATE TrackerHosted)
target_include_directories(vst3-tests PRIVATE "${ROOT}/include/nlohmann-json/include")
target_compile_options(vst3-tests PRIVATE "/FI${ROOT}/windows/cmake/WindowsTestPreamble.hpp")
target_link_options(vst3-tests PRIVATE /STACK:8388608)
add_test(NAME vst3-native COMMAND vst3-tests "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:VST3Fixture>" "${CMAKE_CURRENT_BINARY_DIR}/cache.json")
set_tests_properties(vst3-native PROPERTIES TIMEOUT 30)
foreach(case Lifecycle Editor)
 add_executable(vst3-${case} ${case}Tests.cpp)
 target_link_libraries(vst3-${case} PRIVATE TrackerHosted)
 target_include_directories(vst3-${case} PRIVATE "${ROOT}/include/nlohmann-json/include")
 target_compile_options(vst3-${case} PRIVATE "/FI${ROOT}/windows/cmake/WindowsTestPreamble.hpp")
 target_link_options(vst3-${case} PRIVATE /STACK:8388608)
 add_test(NAME vst3-${case} COMMAND vst3-${case} "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:VST3Fixture>" "${CMAKE_CURRENT_BINARY_DIR}/${case}-cache.json")
 set_tests_properties(vst3-${case} PROPERTIES TIMEOUT 20)
endforeach()
add_executable(vst3-UiOwner UiOwnerTests.cpp)
target_include_directories(vst3-UiOwner PRIVATE "${ROOT}")
target_compile_definitions(vst3-UiOwner PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(vst3-UiOwner PRIVATE user32 ole32)
target_link_options(vst3-UiOwner PRIVATE /STACK:8388608)
add_test(NAME vst3-UiOwner COMMAND vst3-UiOwner)
set_tests_properties(vst3-UiOwner PROPERTIES TIMEOUT 10)
foreach(fault 1 2 3)
 add_library(VST3Fault${fault} MODULE FaultFixture.cpp)
 target_include_directories(VST3Fault${fault} PRIVATE "${SCREAMSEQ_VST3_SDK}")
 target_compile_definitions(VST3Fault${fault} PRIVATE FIXTURE_FAULT=${fault})
 set_target_properties(VST3Fault${fault} PROPERTIES PREFIX "" SUFFIX ".vst3")
endforeach()
add_executable(vst3-Scanner ScannerTests.cpp)
target_link_libraries(vst3-Scanner PRIVATE TrackerHosted)
target_link_options(vst3-Scanner PRIVATE /STACK:8388608)
add_test(NAME vst3-Scanner COMMAND vst3-Scanner "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:VST3Fixture>" "${CMAKE_CURRENT_BINARY_DIR}/scanner-disposable" "$<TARGET_FILE:VST3Fault1>" "$<TARGET_FILE:VST3Fault2>" "$<TARGET_FILE:VST3Fault3>")
set_tests_properties(vst3-Scanner PROPERTIES TIMEOUT 20)
add_executable(vst3-Separate SeparateTests.cpp)
target_link_libraries(vst3-Separate PRIVATE TrackerHosted)
target_include_directories(vst3-Separate PRIVATE "${ROOT}/include/nlohmann-json/include")
target_link_options(vst3-Separate PRIVATE /STACK:8388608)
add_test(NAME vst3-Separate COMMAND vst3-Separate "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:VST3SeparateFixture>" "${CMAKE_CURRENT_BINARY_DIR}/separate-cache.json")
set_tests_properties(vst3-Separate PROPERTIES TIMEOUT 20)
add_executable(vst3-ReviewEditor ReviewEditorTests.cpp)
target_link_libraries(vst3-ReviewEditor PRIVATE TrackerHosted)
target_link_options(vst3-ReviewEditor PRIVATE /STACK:8388608)
foreach(mode 40 41 42)
 add_test(NAME vst3-ReviewEditor-${mode} COMMAND vst3-ReviewEditor "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:VST3Fixture>" "${CMAKE_CURRENT_BINARY_DIR}/editor-${mode}-cache.json" ${mode})
 set_tests_properties(vst3-ReviewEditor-${mode} PROPERTIES TIMEOUT 20)
endforeach()
add_executable(vst3-ReviewSdk ReviewSdkTests.cpp)
target_link_libraries(vst3-ReviewSdk PRIVATE TrackerHosted)
target_include_directories(vst3-ReviewSdk PRIVATE "${ROOT}/include/nlohmann-json/include")
target_link_options(vst3-ReviewSdk PRIVATE /STACK:8388608)
foreach(kind VST3Fixture VST3SeparateFixture)
 foreach(scenario state lifecycle phase catalog)
  add_test(NAME vst3-Review-${kind}-${scenario} COMMAND vst3-ReviewSdk "$<TARGET_FILE:ScreamSeqVST3Scanner>" "$<TARGET_FILE:${kind}>" "${CMAKE_CURRENT_BINARY_DIR}/${kind}-${scenario}-cache.json" ${scenario})
  set_tests_properties(vst3-Review-${kind}-${scenario} PROPERTIES TIMEOUT 20)
 endforeach()
endforeach()
foreach(target VST3Fixture VST3SeparateFixture)
 add_library(${target} MODULE FixtureVST3.cpp ${SCREAMSEQ_VST3_SDK_SOURCES})
 target_include_directories(${target} PRIVATE "${SCREAMSEQ_VST3_SDK}")
 target_compile_definitions(${target} PRIVATE NOMINMAX UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${target} PRIVATE /utf-8)
 target_link_libraries(${target} PRIVATE user32)
 set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".vst3")
 if(target STREQUAL "VST3SeparateFixture")
  target_compile_definitions(${target} PRIVATE FIXTURE_SEPARATE=1)
 endif()
endforeach()
