  # "worker" labels the device-free, desktop-free tests that build.ps1 -Test
  # runs after the portable set. Tests that save with overwrite:false get a
  # fresh private directory from a fixture that runs before them.
  function(screamseq_scratch name)
    set(scratch "${CMAKE_CURRENT_BINARY_DIR}/test-scratch/${name}")
    add_test(NAME ${name}-scratch COMMAND "${CMAKE_COMMAND}" "-DDIRECTORY=${scratch}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/FreshDirectory.cmake")
    set_tests_properties(${name}-scratch PROPERTIES FIXTURES_SETUP ${name}-scratch LABELS "worker;functional;no-realtime-audit" TIMEOUT 60)
  endfunction()
  function(screamseq_worker_test name scratch)
    add_test(NAME ${name} COMMAND ${ARGN})
    set_tests_properties(${name} PROPERTIES FIXTURES_REQUIRED ${scratch}-scratch RESOURCE_LOCK ${scratch}-scratch
      LABELS "worker;functional;no-realtime-audit" TIMEOUT 120)
  endfunction()
